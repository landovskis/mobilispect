11 — Usage

Code Examples

Copy-paste ready snippets. Add the Tailwind config block to unlock all tokens.

<!-- tailwind.config (CDN) -->
<script>
  tailwind.config = {
    theme: {
      extend: {
        fontFamily: {
          display: ['Cormorant', 'serif'],
          sans:    ['Jost', 'sans-serif'],
        },
        colors: {
          cinnabar: { 500: '#C8463A', 600: '#A83530' },
          cream:    { 50:  '#FAFAF7', 100: '#F4F4EF' },
        },
      }
    }
  }
</script>

<!-- Primary button -->
<button class="bg-cinnabar-500 text-white text-sm font-medium
         px-5 py-2.5 rounded-md hover:bg-cinnabar-600 transition-colors">
  Get Started
</button>

<!-- Display heading -->
<h1 class="font-display text-6xl font-light text-cream-900 italic">
  Built with <em class="text-cinnabar-500">intention</em>
</h1>

<!-- Stat card -->
<div class="bg-white rounded-xl border border-cream-200 p-5 shadow-sm">
  <p class="font-mono text-xs text-cream-400 mb-1">On-Time Rate</p>
  <p class="font-display text-5xl font-light"
     style="color: #3D9A6B">87%</p>
</div>
